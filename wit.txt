// ═══════════════════════════════════════════════
function startQuiz(handoutId){
  currentStreak=0;maxStreak=0;setTimeout(()=>updateStreakChip(),50);
  let pool,label;

  if(handoutId==='comprehensive'){
    // Merge all terms for this subject
    const allTermSubjects=SUBJECTS.filter(s=>
      s.code===currentSubject.code &&
      s.year===currentSubject.year &&
      s.sem===currentSubject.sem
    );
    const TERM_ORDER=['prelim','midterm','prefinal','finals'];
    allTermSubjects.sort((a,b)=>TERM_ORDER.indexOf(a.term)-TERM_ORDER.indexOf(b.term));
    pool=allTermSubjects.flatMap(s=>s.handouts.flatMap(h=>h.questions));
    label=currentSubject.code+'  ·  COMPREHENSIVE (ALL TERMS)';
    currentHandout=null;
  } else if(handoutId==='all'){
    pool=currentSubject.handouts.flatMap(h=>h.questions);
    label=currentSubject.code+'  ·  ALL HANDOUTS';
    currentHandout=null;
  } else {
    const h=currentSubject.handouts.find(h=>h.id===handoutId);
    pool=h.questions;
    label=currentSubject.code+'  ·  '+h.label.toUpperCase()+': '+h.title.toUpperCase();
    currentHandout=handoutId;
  }
  questions=shuffle(pool).map(q=>({...q,options:shuffle(q.options)}));
  buildStreakMilestones(questions.length);
  const ALL_MODES=['mc','tf','flash','id'];
  questionModes=questions.map(()=>quizMode==='mix'?ALL_MODES[Math.floor(Math.random()*ALL_MODES.length)]:quizMode);
  current=0;correct=0;wrong=0;answered=false;results=[];questionStates=[];
  document.getElementById('cnt-correct').textContent=0;
  document.getElementById('cnt-wrong').textContent=0;
  document.getElementById('cnt-total').textContent=questions.length;
  document.getElementById('quiz-label').textContent=label;
  document.body.classList.add('in-quiz');
  document.body.classList.remove('showing-result');
  show('counter-bar');
  document.querySelector('.prog-track').style.display='';
  document.getElementById('result-area').classList.add('hidden');
  document.getElementById('quiz-area').classList.remove('hidden');
  const qfb=document.getElementById('quiz-float-bar');if(qfb)qfb.style.display='';
  showSubScreen('screen-quiz');
  renderQuestion();
}

function renderQuestion(){
  answered=questionStates[current]==='correct'||questionStates[current]==='wrong';
  const idx=current,q=questions[idx],mode=questionModes[idx]||'mc';
  document.getElementById('cnt-current').textContent=idx+1;
  const doneCount=questionStates.filter(s=>s==='correct'||s==='wrong').length;
  document.getElementById('progress-fill').style.width=`${(doneCount/questions.length)*100}%`;
  if(mode==='flash')renderFlash(idx,q);
  else if(mode==='tf')renderTF(idx,q);
  else if(mode==='id')renderID(idx,q);
  else renderMC(idx,q);
  updateCirclePanel();
  if(questionStates[idx]==='correct'||questionStates[idx]==='wrong')restoreAnsweredState(idx);
  updatePrevBtn();
  document.getElementById('app-content').scrollTop=0;
  updateFloatBtns();
}

function modeBadge(mode){
  const map={mc:['var(--ok)','var(--ok-dim)','📝 Multiple Choice'],tf:['#7eaee0','rgba(126,174,224,.12)','⚖ True or False'],flash:['var(--skip)','var(--skip-dim)','🃏 Flashcard'],id:['var(--pink-soft)','var(--err-dim)','✍ Identification']};
  const[col,bg,lbl]=map[mode]||map.mc;
  return `<span class="q-mode-badge" style="background:${bg};color:${col};border-color:${col}40">${lbl}</span>`;
}

function nextQuestion(){
  const allAnswered=questionStates.length===questions.length&&questionStates.every(s=>s==='correct'||s==='wrong');
  if(allAnswered){showResult();return}
  const next=findNextUnanswered(current);if(next!==-1){current=next;renderQuestion()}
}

function prevQuestion(){
  let prevIdx=-1;
  for(let i=current-1;i>=0;i--){if(questionStates[i]==='correct'||questionStates[i]==='wrong'||questionStates[i]==='skipped'){prevIdx=i;break}}
  if(prevIdx===-1)for(let i=questions.length-1;i>current;i--){if(questionStates[i]==='correct'||questionStates[i]==='wrong'||questionStates[i]==='skipped'){prevIdx=i;break}}
  if(prevIdx!==-1)jumpTo(prevIdx);
}

function skipQuestion(){
  if(answered)return;
  questionStates[current]='skipped';updateCirclePanel();showToast('⏭  Skipped','wrong');
  const next=findNextUnanswered(current);
  if(next===-1){const fs=questionStates.findIndex(s=>s==='skipped');if(fs!==-1){current=fs;renderQuestion()}return}
  current=next;renderQuestion();document.getElementById('circle-panel').classList.add('visible');
}

function findNextUnanswered(fromIdx){
  for(let i=fromIdx+1;i<questions.length;i++)if(questionStates[i]!=='correct'&&questionStates[i]!=='wrong')return i;
  for(let i=0;i<fromIdx;i++)if(questionStates[i]!=='correct'&&questionStates[i]!=='wrong')return i;
  return -1;
}
function jumpTo(idx){current=idx;renderQuestion()}

function restoreAnsweredState(idx){
  const q=questions[idx],r=results[idx];if(!r)return;
  answered=true;

  // MC
  if(!r.isTF&&!r.isID&&!r.isFlash){
    document.querySelectorAll('.opt-btn').forEach(b=>b.disabled=true);
    const ci=q.options.indexOf(r.userAnswer),cri=q.options.indexOf(q.answer);
    const btns=document.querySelectorAll('.opt-btn');
    if(ci>=0)btns[ci]?.classList.add(r.wasCorrect?'correct':'wrong');
    if(!r.wasCorrect&&cri>=0)btns[cri]?.classList.add('reveal');
  }

  // TF
  if(r.isTF){
    document.querySelectorAll('.opt-btn').forEach(b=>{
      b.disabled=true;
      if(b.dataset.tf===r.userAnswer)b.classList.add(r.wasCorrect?'correct':'wrong');
      if(!r.wasCorrect&&b.dataset.tf===r.correctAnswer)b.classList.add('reveal');
    });
    const pre=document.getElementById('pre-answer-row');
    const fb=document.getElementById('feedback-row');
    if(pre)pre.classList.add('hidden');
    if(fb)fb.classList.remove('hidden');
  }

  // ID
  if(r.isID){
    const inp=document.getElementById('id-input');
    if(inp){inp.value=r.userAnswer==='(gave up)'?'':r.userAnswer;inp.disabled=true;inp.style.borderColor=r.wasCorrect?'var(--ok)':'var(--err)';inp.style.background=r.wasCorrect?'var(--ok-dim)':'var(--err-dim)';}
    const fe=document.getElementById('id-feedback');
    if(fe){
      fe.style.display='block';
      fe.style.background=r.wasCorrect?'var(--ok-dim)':'var(--err-dim)';
      fe.style.border=`1px solid ${r.wasCorrect?'var(--ok-border)':'var(--err-border)'}`;
      fe.style.color=r.wasCorrect?'var(--ok)':'var(--tx)';
      fe.innerHTML=r.wasCorrect?'✓ Correct!':`<span style="color:var(--err)">✗ Your answer:</span> ${escapeHtml(r.userAnswer)}<br><span style="color:var(--ok)">✓ Correct:</span> ${escapeHtml(r.correctAnswer)}`;
    }
    const box=document.getElementById('id-peek-box');
    if(box){box.style.display='block';box.style.borderColor=r.wasCorrect?'var(--ok-border)':'var(--err-border)';box.style.color=r.wasCorrect?'var(--ok)':'var(--err)';box.textContent=r.correctAnswer;}
    ['peek-btn-1','peek-btn-2','peek-btn-3'].forEach(id=>{const b=document.getElementById(id);if(b)b.style.display='none'});
    const sb=document.getElementById('id-submit-btn');if(sb)sb.style.display='none';
    const pre=document.getElementById('pre-answer-row');if(pre)pre.classList.add('hidden');
    const fb=document.getElementById('feedback-row');if(fb)fb.classList.remove('hidden');
  }

  const skipBtnR=document.getElementById('skip-btn');if(skipBtnR)skipBtnR.style.display='none';
  const nb=document.getElementById('next-btn');
  if(nb){const aa=questionStates.length===questions.length&&questionStates.every(s=>s==='correct'||s==='wrong');nb.disabled=false;nb.textContent=aa?'See Results →':'Next →';}
  updatePrevBtn();
}

function updatePrevBtn(){
  const btn = document.getElementById('prev-btn');
  if(!btn) return;
  if(current > 0){
    btn.style.visibility = 'visible';
    btn.textContent = '← Back';
  } else {
    btn.style.visibility = 'hidden';
  }
}

function updateFloatBtns(){
  const left = document.getElementById('sf-left');
  const right = document.getElementById('sf-right');
  if(!left || !right) return;

  const mode = questionModes[current] || 'mc';
  const hasPrev = current > 0;

  // Reset both
  [left, right].forEach(btn => {
    btn.removeAttribute('style');
    btn.className = 'qfb-btn';
    btn.style.display = 'none';
    btn.onclick = null;
  });

  // LEFT — back only if past question 1
  if(hasPrev){
    left.style.display = 'flex';
    left.style.fontSize = '1.5rem';
    left.innerHTML = '‹';
    left.onclick = () => prevQuestion();
  }

  // RIGHT
  if(mode === 'flash'){
    right.style.display = 'flex';
    right.className = 'qfb-btn qfb-next';
    right.style.fontSize = '1.4rem';
    right.innerHTML = '›';
    right.onclick = () => nextFlash();

    } else if(mode === 'tf'){
    if(answered){
      const allDone = questionStates.length === questions.length && questionStates.every(s => s === 'correct' || s === 'wrong');
      right.style.display = 'flex';
      right.className = 'qfb-btn qfb-next';
      right.style.fontSize = allDone ? '.5rem' : '1.5rem';
      right.style.letterSpacing = allDone ? '.05em' : 'normal';
      right.innerHTML = allDone ? 'submit' : '›';
      right.onclick = () => { if(allDone) showResult(); else nextQuestion(); };
    }

  } else if(mode === 'mc'){
    right.style.display = 'flex';
    if(!answered){
      right.className = 'qfb-btn qfb-skip';
      right.style.fontSize = '.56rem';
      right.style.letterSpacing = '.06em';
      right.innerHTML = 'skip';

      right.className = 'qfb-btn qfb-skip';
      right.style.fontSize = '.56rem';
      right.style.letterSpacing = '.06em';

      right.onclick = () => skipQuestion();

function buildStreakMilestones(totalQuestions){
  // How many milestone levels to use (1–10)
  // Always end at 10s; for smaller quizzes use fewer, starting from higher numbers
  const interval = 10; // always trigger every 10 correct answers
  const maxLevels = 10;

  // How many milestones fit in this quiz?
  const count = Math.floor(totalQuestions / interval);
  const levelsToUse = Math.min(count, maxLevels);

  // Start from (10 - levelsToUse + 1) so it always ends at 10
  const startLevel = maxLevels - levelsToUse + 1;

  streakMilestones = [];
  for(let i = 0; i < levelsToUse; i++){
    streakMilestones.push({
      at: (i + 1) * interval,       // trigger at 10, 20, 30...
      level: startLevel + i,         // image level: e.g. 8, 9, 10
      img: `${startLevel + i}s.jpg`
    });
  }

  triggeredStreakMilestones = new Set();
  triggeredWrongMilestones  = new Set();
  consecutiveWrong = 0;
}

function checkStreakMilestone(){
  for(const m of streakMilestones){
    if(currentStreak === m.at && !triggeredStreakMilestones.has(m.at)){
      triggeredStreakMilestones.add(m.at);
      showStreakPopup(m.level, m.img);
      return;
    }
  }
}

function checkWrongMilestone(){
  // Milestones: 5→1wrong, 10→2wrong, 15→3wrong, 20→4wrong
  const wrongLevels = [5,10,15,20];
  for(let i = wrongLevels.length - 1; i >= 0; i--){
    const threshold = wrongLevels[i];
    if(consecutiveWrong === threshold && !triggeredWrongMilestones.has(threshold)){
      triggeredWrongMilestones.add(threshold);
      showWrongPopup(i + 1); // 1wrong, 2wrong, 3wrong, 4wrong
      return;
    }
  }
}

function onCorrectAnswer(){
  currentStreak++;
  if(currentStreak > maxStreak) maxStreak = currentStreak;
  consecutiveWrong = 0;           // reset wrong chain
  triggeredWrongMilestones.clear(); // allow wrong milestones to retrigger fresh
  updateStreakChip();
  checkStreakMilestone();
}

function onWrongAnswer(){
  // Streak cut — only if there was an active streak
  if(currentStreak > 0){
    showStreakCut();
  }
  currentStreak = 0;
  consecutiveWrong++;
  updateStreakChip();
  checkWrongMilestone();
}

function playStreakSound(level){
  const ctx = getAudioCtx();
  // More notes = higher level = more epic
  const noteCount = Math.min(level, 7);
  const baseFreqs = [523,659,784,988,1175,1319,1568];
  const freqs = baseFreqs.slice(0, noteCount);
  freqs.forEach((freq, i) => {
    const osc = ctx.createOscillator(), g = ctx.createGain();
    osc.connect(g); g.connect(ctx.destination);
    osc.frequency.value = freq;
    osc.type = level >= 9 ? 'sine' : level >= 5 ? 'square' : 'triangle';
    g.gain.setValueAtTime(.16, ctx.currentTime + i*.09);
    g.gain.exponentialRampToValueAtTime(.001, ctx.currentTime + i*.09 + .22);
    osc.start(ctx.currentTime + i*.09);
    osc.stop(ctx.currentTime + i*.09 + .25);
  });
}
